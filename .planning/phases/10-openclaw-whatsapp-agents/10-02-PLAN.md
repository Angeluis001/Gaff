Create scripts/verify-cold-steps.ts to check that a cold lead has 5 followup steps scheduled, including the 2 WhatsApp steps at indexes 2 and 3:

```typescript
import { drizzle } from "drizzle-orm/neon-http"
import { neon } from "@neondatabase/serverless"
import { leads, leadFollowupSteps } from "../src/lib/db/schema/index"
import { eq, sql } from "drizzle-orm"

async function main() {
  const db = drizzle(neon(process.env.DATABASE_URL!))

  // Find a cold lead that has followup steps
  const coldLeads = await db
    .select({ id: leads.id, firstName: leads.firstName, classification: leads.classification })
    .from(leads)
    .where(eq(leads.classification, "cold"))
    .limit(5)

  if (coldLeads.length === 0) {
    console.log("No cold leads found. Using scheduleLeadFollowUps to create test steps.")
    console.log("Run: npx tsx scripts/seed-cold-lead-steps.ts to create a cold lead with steps")
    process.exit(0)
  }

  for (const lead of coldLeads) {
    const steps = await db
      .select({ id: leadFollowupSteps.id, stepIndex: leadFollowupSteps.stepIndex, channel: leadFollowupSteps.channel, subject: leadFollowupSteps.subject, dueAt: leadFollowupSteps.dueAt, sentAt: leadFollowupSteps.sentAt })
      .from(leadFollowupSteps)
      .where(eq(leadFollowupSteps.leadId, lead.id))

    console.log(`\nLead: ${lead.firstName} (${lead.id}) — ${steps.length} steps`)
    for (const s of steps.sort((a, b) => (a.stepIndex ?? 0) - (b.stepIndex ?? 0))) {
      console.log(`  stepIndex=${s.stepIndex} channel=${s.channel} subject="${s.subject}" sentAt=${s.sentAt ?? 'pending'}`)
    }

    const whatsappSteps = steps.filter(s => s.channel === 'whatsapp')
    console.log(`  WhatsApp steps: ${whatsappSteps.length} (expected: 2 at stepIndex 2 and 3)`)
  }

  process.exit(0)
}
main().catch(e => { console.error(e); process.exit(1) })
```

Create scripts/seed-completed-booking.ts to insert a booking with status='completed' and date set to yesterday (in UTC, within the Mazatlan window the cron checks):

```typescript
import { drizzle } from "drizzle-orm/neon-http"
import { neon } from "@neondatabase/serverless"
import { bookings, leads, boats } from "../src/lib/db/schema/index"
import { eq } from "drizzle-orm"

async function main() {
  const db = drizzle(neon(process.env.DATABASE_URL!))

  const [lead] = await db
    .select({ id: leads.id, firstName: leads.firstName, phone: leads.phone, whatsappNumber: leads.whatsappNumber })
    .from(leads)
    .where(eq(leads.firstName, "Sarah"))
    .limit(1)

  const [boat] = await db.select({ id: boats.id, name: boats.name }).from(boats).limit(1)

  if (!lead || !boat) {
    console.error("Missing lead or boat — run a seed that creates a Sarah lead first")
    process.exit(1)
  }

  // Yesterday at noon UTC (well within the Mazatlan UTC-7 window)
  const yesterday = new Date()
  yesterday.setDate(yesterday.getDate() - 1)
  yesterday.setHours(12, 0, 0, 0)

  const [booking] = await db
    .insert(bookings)
    .values({
      leadId: lead.id,
      boatId: boat.id,
      date: yesterday,
      tripType: "full_day",
      guests: 4,
      status: "completed",
      totalPrice: "1200.00",
      depositAmount: "600.00",
    })
    .returning()

  console.log("Seeded completed booking:", JSON.stringify({ id: booking.id, date: booking.date, status: booking.status, leadId: booking.leadId }, null, 2))
  console.log("Lead WhatsApp:", lead.whatsappNumber ?? lead.phone)
  process.exit(0)
}
main().catch(e => { console.error(e); process.exit(1) })
```

import { drizzle } from "drizzle-orm/neon-http"
import { neon } from "@neondatabase/serverless"
import { leads, leadFollowupSteps } from "../src/lib/db/schema/index"
import { eq, inArray } from "drizzle-orm"

async function main() {
  const db = drizzle(neon(process.env.DATABASE_URL!))

  // Find leads that have a stepIndex=99 (upsell) followup step
  const upsellSteps = await db
    .select({
      id: leadFollowupSteps.id,
      leadId: leadFollowupSteps.leadId,
      channel: leadFollowupSteps.channel,
      subject: leadFollowupSteps.subject,
      dueAt: leadFollowupSteps.dueAt,
      sentAt: leadFollowupSteps.sentAt,
      createdAt: leadFollowupSteps.createdAt,
    })
    .from(leadFollowupSteps)
    .where(eq(leadFollowupSteps.stepIndex, 99))

  if (upsellSteps.length === 0) {
    console.log("No upsell steps (stepIndex=99) found.")
    console.log("This means either: (a) no real deposits have been processed since the code was deployed,")
    console.log("or (b) the webhook is not creating the step correctly.")
    console.log("\nTo verify the code path, insert a test upsell step manually:")
    console.log("  psql $DATABASE_URL -c \"INSERT INTO lead_followup_steps (lead_id, classification, channel, subject, message, step_index, due_at) SELECT id, 'warm', 'whatsapp', 'Upsell post-booking', 'Test upsell', 99, now() + interval '48 hours' FROM leads LIMIT 1\"")
  } else {
    console.log(`Found ${upsellSteps.length} upsell step(s) (stepIndex=99):`)
    for (const s of upsellSteps) {
      console.log(`  stepId=${s.id} leadId=${s.leadId} channel=${s.channel} dueAt=${s.dueAt} sentAt=${s.sentAt ?? 'pending'}`)
    }
  }

  process.exit(0)
}
main().catch(e => { console.error(e); process.exit(1) })
